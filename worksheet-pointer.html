ใบงาน · พอยน์เตอร์ (Pointer)

ใบงานมี 5 ตอน ไล่จากง่ายไปยาก ใช้ร่วมกับใบความรู้เรื่องพอยน์เตอร์ เมื่อตอบครบแล้วกดส่ง — คำตอบจะถูกส่งไปยังอาจารย์ทางอีเมล

ข้อมูลนักเรียน

ตอนที่ 1 · ไล่โค้ดและทายผลลัพธ์

เขียนผลลัพธ์ที่โปรแกรมแสดงบนหน้าจอ โค้ดทุกข้อถือว่ามี #include <bits/stdc++.h> และ using namespace std; อยู่แล้ว

ข้อ 1.1

int x = 5, y = 8;
int *p = &x;
*p = *p + 3;
p = &y;
*p = *p * 2;
cout << x << " " << y;

ข้อ 1.2

int a = 4;
int *p = &a;
int *q = p;
*q = 10;
cout << a << " " << *p;

ข้อ 1.3

int arr[5] = {3, 6, 9, 12, 15};
int *p = arr + 1;
cout << *p << " " << *(p + 2) << " " << p[1];

ข้อ 1.4

int arr[4] = {1, 2, 3, 4};
for (int *p = arr; p < arr + 4; p++)
    *p = *p * 10;
cout << arr[0] + arr[3];

ข้อ 1.5

void f(int a, int &b, int *c) {
    a++;
    b++;
    (*c)++;
}
int main() {
    int x = 1, y = 1, z = 1;
    f(x, y, &z);
    cout << x << y << z;
}

ข้อ 1.6

int x = 7;
int *p = &x;
int **pp = &p;
**pp = 20;
cout << x << " " << (*pp == &x);

ข้อ 1.7

int arr[3] = {1, 2, 3};
for (int v : arr) v = v * 5;
for (int &v : arr) v = v + 1;
cout << arr[0] << arr[1] << arr[2];

ข้อ 1.8

int arr[6] = {2, 4, 6, 8, 10, 12};
int *p = arr + 1, *q = arr + 5;
cout << q - p << " " << *q - *p;

ตอนที่ 2 · เลือกคำตอบที่ถูกต้อง

เลือกคำตอบที่ถูกต้องที่สุดเพียงข้อเดียว

ข้อ 2.1

ถ้า int x; อยู่ที่ที่อยู่ 0x200 และ int *p = &x; (int ขนาด 4 ไบต์) ค่าของ p + 2 คือข้อใด

ข้อ 2.2

คำสั่ง int* a, b; ข้อใดถูกต้อง

ข้อ 2.3

ข้อใด ไม่ถูกต้อง เกี่ยวกับ reference

ข้อ 2.4

arr[3] มีความหมายเท่ากับข้อใด

ข้อ 2.5

ถ้า p เป็น pointer ชี้ไปที่ struct แล้ว p->score เท่ากับข้อใด

ข้อ 2.6

เหตุใดคำสั่ง scanf("%d", &x); จึงต้องใส่ & หน้า x

ตอนที่ 3 · หาที่ผิดและแก้ไข

แต่ละข้อมีข้อผิดพลาด 1 จุด ให้ระบุบรรทัดที่ผิด อธิบายเหตุผล และเขียนโค้ดที่แก้แล้ว

ข้อ 3.1

int *p;
*p = 10;
cout << *p;

ข้อ 3.2

ต้องการหาผลรวมของสมาชิกทุกตัวในอาร์เรย์

int arr[5] = {1, 2, 3, 4, 5};
int sum = 0;
for (int *p = arr; p <= arr + 5; p++)
    sum += *p;

ข้อ 3.3

ต้องการให้โปรแกรมแสดงผล 15

void addTen(int x) {
    x += 10;
}
int main() {
    int a = 5;
    addTen(a);
    cout << a;
}

ข้อ 3.4

int *getValue() {
    int x = 42;
    return &x;
}
int main() {
    int *p = getValue();
    cout << *p;
}

ตอนที่ 4 · เขียนฟังก์ชัน

เขียนฟังก์ชันตามที่กำหนด และเขียน main เพื่อทดสอบ

ข้อ 4.1

เขียนฟังก์ชันเพิ่มค่าตัวแปรขึ้น 1 จำนวน 2 แบบ

void addOnePtr(int * x); // แบบ pointer
void addOneRef(int & x); // แบบ reference
ตัวอย่าง: int a = 5; addOnePtr(&a); addOneRef(a); แล้ว a มีค่า 7

ข้อ 4.2

เขียนฟังก์ชันหาผลรวมและค่าเฉลี่ยของอาร์เรย์ โดยท่องอาร์เรย์ด้วย pointer

void sumAndAvg(int a[], int n, long long &sum, double &avg);
ตัวอย่าง: a = {4, 7, 1, 8}, n = 4 ได้ sum = 20 และ avg = 5

ข้อ 4.3

เขียนฟังก์ชันนับจำนวนเลขคู่และเลขคี่ในอาร์เรย์

void countEvenOdd(int a[], int n, int &even, int &odd);
ตัวอย่าง: a = {3, 8, 5, 6, 10}, n = 5 ได้ even = 3 และ odd = 2

ข้อ 4.4

เขียนฟังก์ชันเรียงค่าตัวแปร 3 ตัวจากน้อยไปมาก โดยแก้ค่าตัวแปรเดิม

void sort3(int & a, int & b, int & c);
ตัวอย่าง: int x = 9, y = 2, z = 5; sort3(x, y, z); ได้ x = 2, y = 5, z = 9

ตอนที่ 5 · โจทย์ประยุกต์

เขียนโปรแกรมให้สมบูรณ์ ทุกข้อต้องท่องอาร์เรย์ด้วย pointer ห้ามใช้ดัชนี a[i]

ข้อ 5.1 กลับลำดับอาร์เรย์

ใช้ pointer สองตัว ตัวหนึ่งเริ่มจากหัว อีกตัวเริ่มจากท้าย แล้วสลับค่าและขยับเข้าหากัน

  • ข้อมูลเข้า บรรทัดแรก N บรรทัดที่สองเป็นจำนวนเต็ม N ตัว
  • ข้อมูลออก สมาชิกหลังกลับลำดับ คั่นด้วยช่องว่าง
  • ข้อจำกัด 1 ≤ N ≤ 100,000
ข้อมูลเข้าข้อมูลออก
5 1 2 3 4 55 4 3 2 1
4 7 -2 9 00 9 -2 7

ข้อ 5.2 หมุนอาร์เรย์ไปทางซ้าย K ตำแหน่ง

สมาชิกที่หลุดจากด้านซ้ายจะวนไปต่อท้าย

  • ข้อมูลเข้า บรรทัดแรก N และ K บรรทัดที่สองเป็นจำนวนเต็ม N ตัว
  • ข้อมูลออก สมาชิกหลังหมุน คั่นด้วยช่องว่าง
  • ข้อจำกัด 1 ≤ N ≤ 100,000 และ 0 ≤ K ≤ 10^9
ข้อมูลเข้าข้อมูลออก
5 2 1 2 3 4 53 4 5 1 2
3 7 10 20 3020 30 10
คำใบ้: ใช้ฟังก์ชันกลับลำดับจากข้อ 5.1 สามครั้ง และสังเกตว่าหมุน N ครั้งได้อาร์เรย์เดิม

ข้อ 5.3 รวมอาร์เรย์ที่เรียงแล้ว

รวมอาร์เรย์ A และ B ที่เรียงแล้วด้วย pointer สองตัว ห้ามใช้ sort

  • ข้อมูลเข้า N / สมาชิก A / M / สมาชิก B
  • ข้อมูลออก สมาชิกที่รวมแล้ว N + M ตัว คั่นด้วยช่องว่าง
  • ข้อจำกัด 1 ≤ N, M ≤ 100,000
ข้อมูลเข้าข้อมูลออก
3 1 4 7 4 2 3 8 91 2 3 4 7 8 9
2 5 5 3 1 5 61 5 5 5 6

ตรวจชื่อ โรงเรียน และเลขที่ให้ครบก่อนกดส่ง คำตอบจะถูกส่งไปที่อีเมลอาจารย์
